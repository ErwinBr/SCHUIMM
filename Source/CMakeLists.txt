# Library layout (arrows point at dependencies):
#
#   schuimm_core                 Common, Image, Logging, Timing, Input (platform independent)
#   schuimm_platform_interface   <- schuimm_core          (headers only)
#   schuimm_filesystem           <- schuimm_core          (paths and file I/O: POSIX or Win32)
#   schuimm_resource             <- schuimm_filesystem    (assets, CPU-side decoding; libpng private)
#   schuimm_graphics             <- schuimm_core          (interfaces, camera, sprite batching; no graphics API)
#   schuimm_graphics_resources   <- graphics, resource    (texture cache, fonts: assets -> GPU resources)
#   schuimm_audio                <- schuimm_core          (playback, mixer; no audio API)
#   schuimm_audio_miniaudio      <- schuimm_audio         (output device; miniaudio private)
#   schuimm_script_bytecode      <- schuimm_core          (bytecode format, validation; Script/CMakeLists.txt)
#   schuimm_script_compiler      <- script_bytecode       (source -> bytecode; tools and tests only)
#   schuimm_script_runtime       <- script_bytecode       (VM, natives, coroutine scheduler)
#   schuimm_resource_json        <- schuimm_resource      (internal JSON decoding helpers; nlohmann/json)
#   schuimm_animation            <- schuimm_resource      (animation resources, playback, ticks; JSON private)
#   schuimm_navigation           <- schuimm_core          (polygon validation, walkable queries, A* paths)
#   schuimm_room                 <- resource, animation, navigation, script runtime (rooms, actors, movement; JSON private)
#   schuimm_room_graphics        <- room, graphics resources (textures and sprites for rooms)
#   schuimm_adventure_api        <- script_bytecode       (adventure native interface declaration)
#   schuimm_adventure            <- room, graphics, adventure_api (adventure natives, operation completion)
#   schuimm_application          <- interfaces, audio, script runtime (main loop, lifecycle)
#   schuimm_platform             <- platform interface    (exactly one native backend)
#   schuimm_graphics_opengl      <- both interfaces       (OpenGL 3.3 core renderer)
#
# Only the final executable links concrete backends together.

add_subdirectory(Platform)
add_subdirectory(Resource)

add_subdirectory(GraphicsResources)
add_subdirectory(Audio)
add_subdirectory(Script)
add_subdirectory(Animation)
add_subdirectory(Navigation)
add_subdirectory(Room)
add_subdirectory(RoomGraphics)
add_subdirectory(Adventure)
add_subdirectory(Core/Application)
